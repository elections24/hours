function isActiveLocalGovernmentCollationAgent() {
  return request.auth != null
    && exists(/databases/$(database)/documents/electionAgents/$(request.auth.uid))
    && get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.agentRole == 'local_government_collation_agent'
    && get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.active == true
    && get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.status == 'active'
    && get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.geography.lgaCode is string
    && get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.geography.stateCode is string;
}

function canLgaAgentReadWardCollation(data) {
  return isActiveLocalGovernmentCollationAgent()
    && data.recordType == 'ward_collation'
    && data.lgaCode == get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.geography.lgaCode
    && data.stateCode == get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.geography.stateCode
    && data.electionType == get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.electionType;
}

function canLgaAgentReadWardGeography(data) {
  return isActiveLocalGovernmentCollationAgent()
    && data.lgaCode == get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.geography.lgaCode
    && data.stateCode == get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.geography.stateCode;
}

function canLgaAgentCreateCollation(data) {
  return isActiveLocalGovernmentCollationAgent()
    && data.recordType == 'lga_collation'
    && data.createdBy == request.auth.uid
    && data.lgaCode == get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.geography.lgaCode
    && data.stateCode == get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.geography.stateCode
    && data.electionType == get(/databases/$(database)/documents/electionAgents/$(request.auth.uid)).data.electionType
    && data.status == 'submitted';
}

/*
In the existing match /geographyWards/{id} block, extend allow read:
  allow read: if signedIn() || canLgaAgentReadWardGeography(resource.data);
(The signedIn() condition already allows authenticated users in the user's current rules.)

In the existing match /wardCollations/{id} block, extend allow read:
  allow read: if isSuperAdmin()
    || canWardAgentReadWardRecord(resource.data)
    || canLgaAgentReadWardCollation(resource.data);

Add this new collection block:
*/
match /lgaCollations/{id} {
  allow create: if canLgaAgentCreateCollation(request.resource.data);
  allow read: if isSuperAdmin()
    || (request.auth != null && resource.data.createdBy == request.auth.uid);
  allow update, delete: if isSuperAdmin();
}
